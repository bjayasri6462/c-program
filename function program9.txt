#include <stdio.h>


int isPalindrome(int n) {
    int original = n;
    int rev = 0;

    while (n > 0) {
        rev = rev * 10 + (n % 10);
        n = n / 10;
    }

    if (original == rev)
        return 1; 
    else
        return 0; 
}

int main() {
    int N;
    scanf("%d", &N);

    if (isPalindrome(N))
        printf("Palindrome");
    else
        printf("Not Palindrome");

    return 0;
}
